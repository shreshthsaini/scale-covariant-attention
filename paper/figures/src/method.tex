% SCA method figure (Figure 3, fig:method), operator sca_v2. TikZ, self-contained apart from the data and
% images that fig_method_flow.py writes into ../method_flow/ (its docstring and ../method_provenance.json record
% every source, rule and value).
%
% Build from this directory:  ~/.local/bin/tectonic --outdir .. method.tex   (writes ../method.pdf)
% Drawn at print size: 1 unit = 1 pt, canvas 5.5 in x 128 pt, placed at \linewidth. Text: panel titles 8.5 pt
% bold sans, labels 7.5 pt, numbers 7 pt, math 8 pt (scripts 6 pt). Helvetica metrics, Computer Modern math.
%
% Layout (redesign of 2026-11-02, after the round-1 presentation review and the lead's brief):
%   (a) Calibrate at 1024^2, left to right: one calibration image; the key maps of one query at emulated ratio
%       r = 4, to scale on the 64 x 64 grid (full: every key; global: the 16 x 16 block-centre lattice; local:
%       the 16 x 16 window); the closed-form fit J(pi) of one cell with pi* = clip(B/A); a 3 x 3 excerpt of the
%       r = 4 table (dots: the three heads of Figure 5; sigma rising left to right as in Figures 5 and B.1).
%   (b) Sample at 4096^2, left to right: input noise; one head's pi_h at each of the 28 steps (blue: global
%       share, amber: local share; dashed: the gate sigma_g after step 8); the windows of steps 9 to 12 and 13 to 28
%       as nested frames on the real 128-token crop around one query tile; the output.
%   (c) Mix per head, a pale green panel fed by the table: O^glob (blue) and O^loc (amber) into the sum weighted
%       1 - pi_h and pi_h, output O_i (green); bars: the 24 heads of the curve's block at step 20, sorted, amber
%       pi_h under blue 1 - pi_h, the curve's head outlined (green dots link it to the curve).
% Colours (sca_colors.tex): green = SCA (fit, weights, output), amber = local, blue = global, grey = neutral.

% ---------------------------------------------------------------- key map at r = 4, 64 x 64 tokens, to scale
% \KeyMap{x}{y}{side}{content}; content is \KMfull, \KMglobal or \KMlocal, drawn in token coordinates (row down).
% Query tile (2 x 2 tokens at r = 4): tokens 26..27 x 30..31; its 16 x 16 window: rows 19..34, cols 23..38
% (start = clip(2 tau + 1 - 8, 0, 48)); block centres kappa(b) = 4 b + 2.
\newcommand{\KMfull}{\fill[scaFull!55] (0,0) rectangle (64,64);}
\newcommand{\KMglobal}{\foreach \i in {0,...,15}{\foreach \j in {0,...,15}{\fill[scaGlobal] (4*\i+2.5,4*\j+2.5) circle[radius=0.82];}}}
\newcommand{\KMlocal}{\fill[scaLocal] (23,19) rectangle (39,35);}
\newcommand{\KeyMap}[4]{%
  \pgfmathsetmacro{\kmu}{#3/64}%
  \begin{scope}[shift={(#1,#2+#3)}, x=\kmu pt, y=-\kmu pt]
    \fill[white] (0,0) rectangle (64,64);
    #4
    \draw[line width=0.5pt, scaFull] (0,0) rectangle (64,64);
    \fill[scaInk] (29,25) rectangle (33,29);
    \draw[white, line width=0.35pt] (29,25) rectangle (33,29);
  \end{scope}}

% ---------------------------------------------------------------- closed-form fit of one cell
% \FitPlot{x}{y}{w}{h}: J(pi)/E over pi in [0,1], minimum at pi* (green); axes only at the bottom.
\newcommand{\FitPlot}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \pgfmathsetmacro{\fyl}{\FitJlo-0.18*(\FitJhi-\FitJlo)}
    \pgfmathsetmacro{\fyh}{\FitJhi+0.05*(\FitJhi-\FitJlo)}
    \draw[line width=0.5pt, scaMid] (0,0) -- (#3,0);
    \draw[line width=0.5pt, scaMid] (0,0) -- (0,1.5) (#3,0) -- (#3,1.5);
    \draw[line width=1pt, scaInk, domain=0:1, samples=50, smooth]
      plot ({#3*\x}, {#4*((1-2*\FitB*\x+\FitA*\x*\x)-\fyl)/(\fyh-\fyl)});
    \pgfmathsetmacro{\fpy}{#4*(\FitJmin-\fyl)/(\fyh-\fyl)}
    \draw[line width=0.6pt, scaAccent, dash pattern=on 1.2pt off 0.9pt] ({#3*\FitPi},0) -- ({#3*\FitPi},\fpy);
    \fill[scaAccent] ({#3*\FitPi},\fpy) circle[radius=1.5pt];
  \end{scope}}

% ---------------------------------------------------------------- table excerpt, r = 4
% \TableX{x}{y}{cell w}{cell h}: lower-left corner at (x,y); three rows (local, mid, global head), sigma rising.
\newcommand{\HeadDot}[3]{\fill[#3] (#1,#2) circle[radius=1.7pt];}
\newcommand{\TableX}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \i/\dc in {0/scaLocal, 1/scaInk, 2/scaGlobal}{
      \pgfmathsetmacro{\ry}{(2-\i)*#4}
      \HeadDot{-4.5}{\ry+0.5*#4}{\dc}
      \foreach \j in {0,1,2}{
        \fill[tab\i\j] (\j*#3, \ry) rectangle ++(#3,#4);
        \node at ({(\j+0.5)*#3}, {\ry+0.5*#4}) {\sml\csname tabv\i\j\endcsname};
      }
    }
    \draw[white, line width=0.8pt] (0,0) grid[xstep=#3, ystep=#4] (3*#3, 3*#4);
    \draw[scaAccent, line width=0.9pt] (#3, 0) rectangle ++(#3, #4);
    \foreach \s [count=\j from 0] in \TabSigmas{\node[anchor=south, text=scaMid] at ({(\j+0.5)*#3}, 3*#4+1.2) {\sml \s};}
    \node[anchor=south east] at (-1.5, 3*#4+1.0) {\mth $\sigma$};
  \end{scope}}

% ---------------------------------------------------------------- pi_h of one head over the 28 steps
% \PiPlot{x}{y}{w}{h}: plot box lower-left at (x,y). Blue: global share, amber: local share pi_h.
\newcommand{\PiPlot}[4]{%
  \pgfmathsetmacro{\ppu}{#3/28}%
  \begin{scope}[shift={(#1,#2)}, x=\ppu pt, y=#4pt]
    \fill[scaGlobalLight] (0.5,0) rectangle (28.5,1);
    \fill[scaLocalLight] (0.5,0) -- \PiStair -- (28.5,0) -- cycle;
    \draw[line width=1.1pt, scaAccent, line join=round] \PiStair;
    \draw[line width=0.5pt, scaMid] (0.5,1) -- (0.5,0) -- (28.5,0);
    \foreach \t in {1,9,13,28}{\draw[line width=0.5pt, scaMid] (\t,0) -- (\t,-0.06);
      \node[anchor=north, text=scaMid] at (\t,-0.09) {\sml\t};}
    \foreach \v in {0,1}{\draw[line width=0.5pt, scaMid] (0.5,\v) -- (0.05,\v);}
    \node[anchor=east, text=scaMid] at (0.0,0) {\sml 0};
    \node[anchor=east, text=scaMid] at (0.0,1) {\sml 1};
    \draw[line width=0.6pt, scaInk, dash pattern=on 1.6pt off 1.1pt] (8.5,0) -- (8.5,1);
    \node[anchor=north west] at (8.85,0.97) {\mth $\sigma_g$};
    \fill[white] (\PiDotStep,\PiDotVal) circle[radius=2.3pt];
    \fill[scaAccent] (\PiDotStep,\PiDotVal) circle[radius=1.5pt];
  \end{scope}}

% ---------------------------------------------------------------- nested windows on the real crop
% \Frames{x}{y}{side}: crop = the w = 128 window; frames w = 128, 112, 96, 80, 64, step tabs on the left edges.
\newcommand{\Frames}[3]{%
  \begin{scope}[shift={(#1,#2)}]
    \node[anchor=south west] at (0,0) {\includegraphics[width=#3pt]{../method_flow/crop.jpg}};
    \foreach \o [count=\k from 0] in \FrameOffsets{
      \draw[white, line width=1.0pt] ({#3*\o},{#3*\o}) rectangle ({#3*(1-\o)},{#3*(1-\o)});
      \draw[scaLocal, line width=0.55pt] ({#3*\o},{#3*\o}) rectangle ({#3*(1-\o)},{#3*(1-\o)});
    }
    \fill[scaInk] ({#3*\QueryCol},{#3*(1-\QueryRow-\QuerySide)}) rectangle ++({#3*\QuerySide},{#3*\QuerySide});
    \draw[white, line width=0.4pt] ({#3*\QueryCol},{#3*(1-\QueryRow-\QuerySide)}) rectangle ++({#3*\QuerySide},{#3*\QuerySide});
    \node[anchor=north west, fill=white, inner xsep=1pt, inner ysep=0.8pt] at (0.6,{#3-0.6}) {\sml 9};
    \node[anchor=south west, fill=white, inner xsep=1pt, inner ysep=0.8pt] at ({#3*0.25+0.5},{#3*0.25+0.5}) {\sml 13+};
  \end{scope}}

% ---------------------------------------------------------------- stacked bars: 24 heads of one block
% \Bars{x}{y}{w}{h}: amber pi_h at the bottom, blue 1 - pi_h on top; the head of the curve outlined.
\newcommand{\Bars}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \pgfmathsetmacro{\bw}{#3/24}
    \foreach \v/\hi [count=\k from 0] in \PiBars{
      \fill[scaLocal] ({\k*\bw+0.25}, 0) rectangle ({(\k+1)*\bw-0.25}, {#4*\v});
      \fill[scaGlobal!85] ({\k*\bw+0.25}, {#4*\v}) rectangle ({(\k+1)*\bw-0.25}, #4);
      \ifnum\hi=1
        \draw[scaAccent, line width=0.8pt] ({\k*\bw-0.25}, -0.6) rectangle ({(\k+1)*\bw+0.25}, {#4+0.6});
        \fill[white] ({(\k+0.5)*\bw}, {#4+3.4}) circle[radius=2.3pt];
        \fill[scaAccent] ({(\k+0.5)*\bw}, {#4+3.4}) circle[radius=1.5pt];
      \fi
    }
  \end{scope}}
