% SCA method diagram (main.tex Figure 3, fig:method), TikZ, self-contained.
%
% Build from this directory:  ~/.local/bin/tectonic --outdir .. method.tex   (writes ../method.pdf)
% Assets and data macros live in ../method_flow/ and are written by
% fig_method_flow.py (its docstring and ../method_provenance.json record every
% source). Full text width (5.5 in) and placed at 100%, so sizes here are
% printed sizes: sans labels 7.5 to 9 pt (Helvetica metrics), math in Computer
% Modern at 9 pt with 7.2 pt scripts, so no glyph is below 7 pt. Titles have
% their own row; the 28-step strip sits below the (a) title and ends at the
% output's right edge, clear of panel (b).
%
% Colour meanings (sca_colors.tex): blue = global branch, amber = local
% branch, green = SCA's mix and output, grey = neutral. The pi colormap runs
% from blue (pi = 0, global) through a light neutral to amber (pi = 1, local).
%
% (a) Sample at 4096^2, left to right: the input latent x_sigma (a real SCA
%     output, forward-noised for illustration) with a schematic token grid
%     (s = 4, n0 = 8, t = 2, so 32 x 32 tokens): blue dots are the block-centre
%     tokens kappa(b) = s b + floor(s/2), the amber square the native n0 x n0
%     window of the black t x t query tile. The global branch keys on the
%     n0 x n0 grid of block centres (query at its coarse cell c(p)); the local
%     branch on the window at full resolution. Each head h mixes them with
%     pi_h(sigma), the chips (24 heads of block 8 at the drawn step), and the
%     output is the clean 4096^2 sample. Header strip: the 28 steps, the first
%     8 (sigma > sigma_g = 0.89) global only, then each step split into the
%     mean local share (amber) and global share (blue); the drawn step is
%     outlined in green. The local view sits level with the window, joined by
%     dashed zoom lines; the global grid is fed by the whole image.
% (b) Calibrate at 1024^2: full attention and the two operators at emulated
%     scale s_c = 4 on the native grid (16 x 16 schematic) fit the table of
%     block 8 (24 heads x 14 noise bins, sigma 1 -> 0 left to right); sigma >
%     sigma_g is veiled, and the outlined column is the bin read in (a).

\pgfmathsetmacro{\cw}{(\sxb-\sxa)/28}

  \pgfmathsetmacro{\xb}{\xa+\cw}
  \ifnum\k>\nglobal
    \pgfmathsetmacro{\ys}{\sya+(1-\p)*(\syb-\sya)}
    \fill[scaGlobal] (\xa,\sya) rectangle (\xb,\ys);
    \fill[scaLocal]  (\xa,\ys) rectangle (\xb,\syb);
  \else
    \fill[scaGlobal] (\xa,\sya) rectangle (\xb,\syb);
  \fi
}
\foreach \k in {1,...,27} { \draw[white, line width=0.35pt] (\sxa+\k*\cw,\sya) -- (\sxa+\k*\cw,\syb); }
\node[lbl, text=white, anchor=center, fill=scaGlobal, inner xsep=1.5pt, inner ysep=0.5pt] at (\sxa+0.5*\nglobal*\cw, 0.5*\sya+0.5*\syb) {global only};
\pgfmathsetmacro{\xg}{\sxa+\nglobal*\cw}
\draw[scaInk, line width=0.6pt] (\xg,\sya-0.06) -- (\xg,\syb+0.04);
\node[sym, anchor=north west] at (\xg+0.06, \sya-0.03) {$\sigma_g$};
% drawn step (green outline, like the bin read in (b))

\draw[scaAccent, line width=0.8pt] (\sxa+\curstep*\cw-\cw,\sya-0.02) rectangle (\sxa+\curstep*\cw,\syb+0.02);

% ================================================================ (a) flow

  \foreach \i in {4,8,...,28} { \draw[white, opacity=0.85, line width=0.3pt] (\i,0) -- (\i,32) (0,\i) -- (32,\i); }
  \fill[scaLocal, opacity=0.22] (13,7) rectangle (21,15);
  \draw[halo] (13,7) rectangle (21,15);

  \coordinate (sref) at (4.5,6.5);
  \foreach \i in {2,6,...,30} { \foreach \j in {2,6,...,30} {
    \fill[white] (\i+0.5,\j+0.5) circle[radius=1.35pt];

  } }
  \fill[white] (15.75,9.75) rectangle (18.25,12.25);
  \fill[scaInk] (16,10) rectangle (18,12);

\node[sym, anchor=south, fill=white, inner sep=0.6pt, rounded corners=1pt] at ($(sref)+(0,0.06)$) {$s$};

  \draw[scaLocal, line width=0.6pt, dash pattern=on 2pt off 1.2pt] (\wxr,\ya) -- (\VX,\yb);
